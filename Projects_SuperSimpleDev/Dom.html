Method 1 — Inline onclick + if / else

Unique: onclick is written directly inside the HTML button.

Working Demo

HTML


<button
    id="newsletterBtn"
    onclick="toggleNewsletter()">
    Subscribe
</button>

JavaScript


function toggleNewsletter() {

    const newsletterBtn =
        document.getElementById('newsletterBtn');

    if (newsletterBtn.textContent.trim() === 'Subscribe') {

        newsletterBtn.textContent = 'Unsubscribe';

    } else {

        newsletterBtn.textContent = 'Subscribe';

    }
}

Method 2 — addEventListener() + if / else

Unique: The click event is connected in JavaScript using addEventListener().

Working Demo

HTML


<button id="notificationBtn">
    Subscribe
</button>

JavaScript


function toggleNotifications() {

    if (
        notificationBtn.textContent.trim()
        === 'Subscribe'
    ) {

        notificationBtn.textContent = 'Subscribed';

    } else {

        notificationBtn.textContent = 'Subscribe';

    }
}


const notificationBtn =
    document.getElementById('notificationBtn');


notificationBtn.addEventListener(
    'click',
    toggleNotifications
);

Method 4 — getElementsByClassName() + [0]

Unique: getElementsByClassName() can find multiple elements, so [0] gets the first matching element.

Working Demo

HTML


<button
    class="akatsuki-btn"
    onclick="main()">
    Hak
</button>

JavaScript


const AkatsukiBtn =
    document.getElementsByClassName(
        'akatsuki-btn'
    )[0];


function main() {

    if (
        AkatsukiBtn.textContent.trim()
        === 'Hak'
    ) {

        AkatsukiBtn.textContent = 'Yona';

    } else {

        AkatsukiBtn.textContent = 'Hak';

    }
}

Method 5 — Input + Enter key + nested ternary

Unique: Uses input.value, Number(), Number.isNaN(), keydown, event.key, and nested ternary.

Working Demo

HTML


<input
    type="text"
    class="user-input"
    onkeydown="handleKeyDown(event)"
    placeholder="Cost of order">

<button onclick="calculateTotal()">
    Calculate
</button>

<p class="display-Cost"></p>

JavaScript


function handleKeyDown(event) {

    if (event.key === 'Enter') {
        calculateTotal();
    }

}


function calculateTotal() {

    const userInput =
        document.querySelector('.user-input');

    const cost =
        Number(userInput.value);


    const result =
        Number.isNaN(cost)
            ? 'Not a number'
            : cost > 40
                ? `$${cost + 10}`
                : `$${cost}`;


    document.querySelector('.display-Cost')
        .textContent = result;

}
